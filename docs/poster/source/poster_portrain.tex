    \item \textbf{\color{blue}Challenge:} Machine Unlearning (MUL) is crucial for data governance and privacy compliance.

    \begin{itemize}

    \item \textbf{\color{blue}VLM-Specific

    \item \textbf{\color{blue} Language as Retain-Free Solution for VLMs.} Use {text prototypes as a retain-free mechanism} to \textbf{\color{ForestGreen}guide unlearning} - no need for image–text retain pairs.
    \vspace{0.5em}
 
    \textbf{\color{blue}Key Contributions:}
     
    \begin{itemize}
    
        \item We propose {{DIET}}: A {\textbf{principled, retain-data-free}} unlearning method for VLMs.
        \vspace{0.5em}
        \item Unlearning by achieving \textbf{asymptotically negligible influence} of forget data via \textbf{hyperbolic geometry} (Poincaré Ball Model)
        \vspace{0.5em}
        \textbf{\color{BrickRed}effectively PUSHING the concept to infinity.}

    \end{itemize}

\headerbox{\bf\color{blue}  The Geometry of Forgetfulness}{name=formulation,column=3,span=2}{
    % We leverage the geometry of hyperbolic space to achieve asymptotic unlearning.
\vspace{0.5em}


\textbf{\color{blue}Hyperbolic Space for Retain-Free Unlearning:
}    \begin{itemize}
        \item \textbf{Goal:} 
        % Push {``{forget}''} embeddings to \textbf{infinity} (out of model's reach).
        To  {``{forget}''}, the model must push unwanted embeddings far away-ideally to \textbf{infinity}, where they have no semantic influence.

        \item \textbf{Euclidean ($\mathbb{R}^n$)}: ``{Infinity}'' is ill-defined; no tractable distance measure to infinity.
        \item \textbf{\color{blue}Poincar\'e Ball Model $(\mathbb{B}^n)$:}
The canonical model of Hyperbolic space ($\mathbb{H}^n), \mathbb{B}^n = \{\mathbf{z} \in \mathbb{R}^n : \|\bmrm{z}\| < 1\}$ where the unit sphere ($\partial\mathbb{B}$) serves as the \textbf{ideal boundary, representing hyperbolic infinity}.
    \end{itemize}
    \vspace{0.5em}
    
  

\vspace{0.5em}
  % \textbf{\color{blue}Busemann Function ($\delta_{\bmrm{p}}(\bmrm{z})$):}
  
  % Measures the directional distance to hyperbolic infinity along the geodesic path $\gamma_{\bmrm{p}}$ connecting $\bmrm{z} \in \mathbb{B}^n$  to an ideal boundary point $\bmrm{p} \in \partial\mathbb{B}^n$.


% \begin{equation}
%     \delta_{\bmrm{p}}(\bmrm{z}) = \lim_{t\to\infty}(d_{\mathbb{B}}(\gamma_{\bmrm{p}}(t), \bmrm{z}) - t),
%     \label{eqn:busemann}
% \end{equation}
% where $d_{\mathbb{B}}\big(\gamma_{\bmrm{p}}(0),\,\gamma_{\bmrm{p}}(t)\big)=t$.

% % \textbf{\color{blue} Intuition:} It quantifies `{how far}' an embedding $\bmrm{z}$ is from effectively reaching the infinite boundary in the direction of $\bmrm{p}$.




\textbf{\color{blue}Busemann Function ($\delta_{\bmrm{p}}(\bmrm{z})$):}
  Measures the {hyperbolic distance} from  $\bmrm{z} \in \mathbb{B}^n$ to points at the boundary $\bmrm{p} \in \partial\mathbb{B}^n$.
  


\begin{equation}
    \textcolor{myviolet}{\delta_{\bmrm{p}}(\bmrm{z})} =
    \textcolor{myred}{\lim_{t\to\infty}}
    \left(
        \textcolor{myorange}{d_{\mathbb{B}}(\textcolor{myteal}{\gamma_{\bmrm{p}}(t)}, \textcolor{myblue}{\bmrm{z}})}
        - \textcolor{Magenta}{t}
    \right)
\end{equation}
The \textbf{\textcolor{myviolet}{Busemann distance}} tells us, \textbf{\textcolor{myred}{at the infinite boundary}}, how much ``{excess}'' \textbf{\textcolor{myorange}{hyperbolic distance}} remains between an \textbf{\textcolor{myblue}{interior point $\in \mathbb{B}^n$} } and its \textbf{\textcolor{myteal}{ideal path to infinity }}, after normalizing for the \textbf{\textcolor{RubineRed}{path length already traversed}}.

\vspace{1em}
 \textbf{\color{blue}Unlearning Action:} Minimizing $\delta_p(\bmrm{z})$ forcefully drives the forget embedding $\bmrm{z}$ along a geodesic path \textbf{directly towards $\bmrm{p}$ at infinity ($\partial\mathbb{B}$).}
    
}

\headerbox{\bf\color{blue} The DIET Framework}{name=method,column=0,below=contribution,span=5}{
\begin{minipage}[t]{0.52\textwidth}



%     \textbf{\color{blue}Geodesic Transport:}
% Leverages \textbf{Busemann distance} to guide forget embeddings along \textbf{hyperbolic geodesics} towards the \textbf{boundar}y ($\partial\mathbb{B}$, i.e., infinity).
% \vspace{0.5em}

    \textbf{\color{blue}Semantic Erasure}
    As embeddings approach the boundary, \textbf{their influence becomes infinitely distant} from interior points, effectively \textbf{erasing semantic contribution}.

    \vspace{0.5em}
    \begin{center}
     \includegraphics[width=\textwidth]{AAAI2026/main_technical_conference_9407@AAAI2026/images/diet_arch.pdf}
    \end{center}
    \vspace{-0.8em}
    

    % \textbf{\color{blue}Guiding the Forgetting:} 
    % \begin{itemize}
    %     \item Optimal Transport (OT) finds an \textbf{optimal, adaptive assignment} between forget image embeddings and the set of mismatched text prototypes.
    %     \item \textbf{Busemann function} guides forget embeddings along \textbf{hyperbolic geodesics} towards the boundary.
    % \end{itemize}
    \textbf{\color{blue}Guiding the Forgetting:}
\begin{itemize}
    \item \textit{\color{blue}Target Direction (Language):} Text prototypes (our retain-free signal) \textbf{define the ideal boundary point} $\bmrm{p}$ (infinity) for the concept to be forgotten.
    \item \textit{\color{blue}Optimal Transport (OT):} Finds \textbf{an optimal, adaptive assignment} between forget image embeddings and the set of mismatched text prototypes. This initiates the push.
    \item \textit{\color{blue}Busemann Function:} Actively steers forget embeddings along the defined \textbf{hyperbolic geodesics} towards the boundary, based on the OT assignment.
    
\end{itemize}
